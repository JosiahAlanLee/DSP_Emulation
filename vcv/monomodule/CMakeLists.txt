cmake_minimum_required(VERSION 3.22)
# The released Mac packages target macOS 11 or newer, as Machinemodule's do.
if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND NOT DEFINED ENV{MACOSX_DEPLOYMENT_TARGET})
    set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
endif()
project(MonomoduleRack LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
# shnolk's Monomodule (AGPL-3.0) and the dsp56300 revision it patches, as tools/fetch-core.sh fetches them.
set(MONOMODULE_COMMIT 23e541021b92d1a5978f36c0a552ac443376df1a)
set(CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/_deps/monomodule" CACHE PATH "Pinned Monomodule checkout")
set(DSP56300_DIR "${CMAKE_CURRENT_SOURCE_DIR}/_deps/dsp56300" CACHE PATH "dsp56300 at Monomodule's commit, with its patch applied")
set(RACK_DIR "" CACHE PATH "VCV Rack 2 SDK")
find_package(Threads REQUIRED)
if(EXISTS "${CORE_DIR}/MONOMODULE_REVISION")
    file(READ "${CORE_DIR}/MONOMODULE_REVISION" CORE_HEAD)
    string(STRIP "${CORE_HEAD}" CORE_HEAD)
    set(CORE_RESULT 0)
else()
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CORE_DIR}" OUTPUT_VARIABLE CORE_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE CORE_RESULT)
endif()
if(NOT CORE_RESULT EQUAL 0 OR NOT CORE_HEAD STREQUAL MONOMODULE_COMMIT)
    message(FATAL_ERROR "Monomodule core must be ${MONOMODULE_COMMIT}; run tools/fetch-core.sh")
endif()

if(MINGW)
    include(cmake/mingw.cmake)
endif()
# The emulator's static libraries, as Monomodule's own build adds them, from the patched checkout.
set(MNM_DSP56300_DIR "${DSP56300_DIR}" CACHE PATH "" FORCE)
set(MNM_BUILD_TESTS OFF)
include("${CORE_DIR}/cmake/dsp56300.cmake")
if(MINGW)
    # Expose declarations for APIs which MmuHelper already resolves dynamically, as Machinemodule does.
    target_compile_definitions(dsp56kBase PRIVATE _WIN32_WINNT=0x0a00 NTDDI_VERSION=0x0a000005)
endif()

# The parts of Monomodule's engine a Rack voice needs: the OS decoder, the DSP harness and the host model.
# The resampler comes through the shared adapter's bridge, compiled once. The pinned headers use size_t
# without <cstddef>; pre-include it rather than change them.
set(MNM_CORE "${CORE_DIR}/src/core")
add_library(mnmvoice STATIC
    ${MNM_CORE}/firmware/Firmware.cpp ${MNM_CORE}/dsp/DspEngine.cpp ${MNM_CORE}/dsp/Digibank.cpp
    ${MNM_CORE}/host/HostModel.cpp ${MNM_CORE}/host/Lfo.cpp ${MNM_CORE}/MonoVoice.cpp)
target_include_directories(mnmvoice PUBLIC "${MNM_CORE}")
target_link_libraries(mnmvoice PUBLIC dsp56kEmu)
if(MSVC)
    target_compile_options(mnmvoice PUBLIC /FIstddef.h)
else()
    target_compile_options(mnmvoice PUBLIC -include stddef.h)
endif()

# The shared adapter (vcv/adapter, AGPL-3.0-only): worker, queues, renderer, sound-link receiver.
add_library(rack_adapter STATIC src/engine/Worker.hpp ../adapter/src/ResamplerBridge.cpp)
target_include_directories(rack_adapter PUBLIC src/engine "${MNM_CORE}" "${MNM_CORE}/dsp")
target_link_libraries(rack_adapter PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(rack_adapter PRIVATE -Wall -Wextra -fno-fast-math)
endif()

enable_testing()
add_executable(renderer_test tests/renderer_test.cpp)
target_link_libraries(renderer_test PRIVATE rack_adapter)
add_test(NAME renderer COMMAND renderer_test)
add_executable(firmware_test tests/firmware_test.cpp)
target_link_libraries(firmware_test PRIVATE rack_adapter mnmvoice)
if(RACK_DIR)
    find_library(RACK_LIBRARY NAMES Rack PATHS "${RACK_DIR}" NO_DEFAULT_PATH REQUIRED)
    add_library(plugin SHARED src/plugin.cpp src/Monomodule.cpp)
    if(MINGW)
        target_sources(plugin PRIVATE src/plugin.def)
        target_link_options(plugin PRIVATE -Wl,--exclude-all-symbols)
    endif()
    target_include_directories(plugin PRIVATE "${RACK_DIR}/include" "${RACK_DIR}/dep/include")
    target_link_libraries(plugin PRIVATE rack_adapter mnmvoice "${RACK_LIBRARY}" ${CMAKE_DL_LIBS})
    set_target_properties(plugin PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    if(UNIX AND NOT APPLE)
        target_compile_options(plugin PRIVATE -fno-gnu-unique -fno-fast-math)
        target_link_options(plugin PRIVATE -static-libstdc++ -static-libgcc "-Wl,-rpath=/tmp/Rack2")
    endif()
    if(APPLE)
        target_link_options(plugin PRIVATE "-undefined" "dynamic_lookup")
    endif()
    if(UNIX)
        set_target_properties(plugin PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "/tmp/Rack2")
    endif()
    add_executable(module_test tests/module_test.cpp)
    target_include_directories(module_test PRIVATE "${RACK_DIR}/include" "${RACK_DIR}/dep/include")
    target_link_libraries(module_test PRIVATE rack_adapter mnmvoice "${RACK_LIBRARY}" ${CMAKE_DL_LIBS})
    if(WIN32)
        target_link_libraries(module_test PRIVATE ws2_32)
    endif()
    if(UNIX AND NOT APPLE)
        target_link_options(module_test PRIVATE -Wl,--allow-shlib-undefined)
    endif()
    set_target_properties(module_test PROPERTIES BUILD_RPATH "${RACK_DIR}")
    set(GORGON_DIR "" CACHE PATH "Optional matching Gorgon checkout for real module interop")
    if(GORGON_DIR)
        target_compile_definitions(module_test PRIVATE GORGON_SOURCE="${GORGON_DIR}/rack/src/Gorgon.cpp")
    endif()
    add_test(NAME rack_module COMMAND module_test)
    add_executable(plugin_load_test tests/plugin_load_test.cpp)
    target_include_directories(plugin_load_test PRIVATE "${RACK_DIR}/include" "${RACK_DIR}/dep/include")
    target_link_libraries(plugin_load_test PRIVATE "${RACK_LIBRARY}" ${CMAKE_DL_LIBS})
    set_target_properties(plugin_load_test PROPERTIES BUILD_RPATH "${RACK_DIR}")
    if(UNIX AND NOT APPLE)
        target_link_options(plugin_load_test PRIVATE -Wl,--allow-shlib-undefined)
    endif()
    add_dependencies(plugin_load_test plugin)
    add_test(NAME plugin_load COMMAND plugin_load_test $<TARGET_FILE:plugin>)
    if(WIN32 OR CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64" OR
       (NOT CMAKE_OSX_ARCHITECTURES AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$"))
        foreach(target plugin module_test plugin_load_test)
            target_compile_options(${target} PRIVATE -march=nehalem)
        endforeach()
    endif()
endif()
if(MINGW)
    foreach(target plugin module_test plugin_load_test renderer_test firmware_test)
        if(TARGET ${target})
            target_link_options(${target} PRIVATE -static -static-libstdc++ -static-libgcc)
        endif()
    endforeach()
endif()
foreach(target renderer_test firmware_test module_test plugin_load_test)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
    endif()
endforeach()
option(MNM_RACK_SANITIZE "Sanitize the adapter and its test (not the emulator)" OFF)
if(MNM_RACK_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target rack_adapter renderer_test)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()
