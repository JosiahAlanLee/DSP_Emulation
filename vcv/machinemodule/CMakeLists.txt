cmake_minimum_required(VERSION 3.22)
project(MachinemoduleRack LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Older pinned dependencies still declare CMake 3.5 compatibility.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(MACHINEMODULE_COMMIT a2b6cf2478cbe94dfe30bc42da39e221df0e2556)
set(CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/_deps/machinemodule" CACHE PATH "Pinned Machinemodule checkout")
set(RACK_DIR "" CACHE PATH "VCV Rack 2 SDK")
find_package(Threads REQUIRED)
if(EXISTS "${CORE_DIR}/MACHINEMODULE_REVISION")
    file(READ "${CORE_DIR}/MACHINEMODULE_REVISION" CORE_HEAD)
    string(STRIP "${CORE_HEAD}" CORE_HEAD)
    set(CORE_RESULT 0)
else()
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CORE_DIR}" OUTPUT_VARIABLE CORE_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE CORE_RESULT)
endif()
if(NOT CORE_RESULT EQUAL 0 OR NOT CORE_HEAD STREQUAL MACHINEMODULE_COMMIT)
    message(FATAL_ERROR "Machinemodule core must be ${MACHINEMODULE_COMMIT}; run tools/fetch-core.sh")
endif()
if(MINGW)
    include(cmake/mingw.cmake)
endif()
add_subdirectory("${CORE_DIR}" upstream EXCLUDE_FROM_ALL)
if(MINGW)
    # Expose declarations for APIs which MmuHelper already resolves dynamically.
    target_compile_definitions(dsp56kBase PRIVATE _WIN32_WINNT=0x0a00 NTDDI_VERSION=0x0a000005)
    # The 68k dependency also spells this SDK include with a capital W.
    set(MC68K_DIR "${CORE_DIR}/libs/gearmulator-md-mm/source/mc68k")
    file(READ "${MC68K_DIR}/logging.cpp" text)
    string(REPLACE "<Windows.h>" "<windows.h>" text "${text}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mc68k-logging.cpp" "${text}")
    get_target_property(sources 68kEmu SOURCES)
    list(REMOVE_ITEM sources logging.cpp)
    list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/mc68k-logging.cpp")
    set_property(TARGET 68kEmu PROPERTY SOURCES "${sources}")
    target_include_directories(68kEmu PRIVATE "${MC68K_DIR}")
endif()
add_library(rack_adapter STATIC src/engine/Worker.hpp src/engine/ResamplerBridge.cpp)
target_include_directories(rack_adapter PUBLIC src/engine "${CORE_DIR}/desktop/src/core")
target_link_libraries(rack_adapter PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(rack_adapter PRIVATE -Wall -Wextra -fno-fast-math)
endif()
enable_testing()
add_executable(control_test tests/control_test.cpp)
target_link_libraries(control_test PRIVATE rack_adapter)
add_test(NAME control COMMAND control_test)
add_executable(renderer_test tests/renderer_test.cpp)
target_link_libraries(renderer_test PRIVATE rack_adapter)
add_test(NAME renderer COMMAND renderer_test)
add_executable(worker_test tests/worker_test.cpp)
target_link_libraries(worker_test PRIVATE rack_adapter)
add_test(NAME worker COMMAND worker_test)
add_executable(firmware_test tests/firmware_test.cpp)
target_link_libraries(firmware_test PRIVATE rack_adapter mdcore ${CMAKE_DL_LIBS})
add_executable(jit_test tests/jit_test.cpp)
target_link_libraries(jit_test PRIVATE mdcore Threads::Threads ${CMAKE_DL_LIBS})
add_test(NAME dsp_jit COMMAND jit_test)
if(RACK_DIR)
    find_library(RACK_LIBRARY NAMES Rack PATHS "${RACK_DIR}" NO_DEFAULT_PATH REQUIRED)
    add_library(plugin SHARED src/plugin.cpp src/Machinemodule.cpp)
    if(MINGW)
        # Automatic exports otherwise include LTO-internal emulator symbols.
        target_sources(plugin PRIVATE src/plugin.def)
        target_link_options(plugin PRIVATE -Wl,--exclude-all-symbols)
    endif()
    target_include_directories(plugin PRIVATE "${RACK_DIR}/include" "${RACK_DIR}/dep/include")
    target_link_libraries(plugin PRIVATE rack_adapter mdcore "${RACK_LIBRARY}" ${CMAKE_DL_LIBS})
    set_target_properties(plugin PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    if(UNIX AND NOT APPLE)
        target_compile_options(plugin PRIVATE -fno-gnu-unique -fno-fast-math)
        target_link_options(plugin PRIVATE -static-libstdc++ -static-libgcc "-Wl,-rpath=/tmp/Rack2")
    endif()
    if(APPLE)
        target_link_options(plugin PRIVATE "-undefined" "dynamic_lookup")
    endif()
    if(UNIX)
        set_target_properties(plugin PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "/tmp/Rack2")
    endif()
    add_executable(module_test tests/module_test.cpp)
    target_include_directories(module_test PRIVATE "${RACK_DIR}/include" "${RACK_DIR}/dep/include")
    target_link_libraries(module_test PRIVATE rack_adapter mdcore "${RACK_LIBRARY}" ${CMAKE_DL_LIBS})
    if(WIN32)
        target_link_libraries(module_test PRIVATE ws2_32)
    endif()
    if(UNIX AND NOT APPLE)
        target_link_options(module_test PRIVATE -Wl,--allow-shlib-undefined)
    endif()
    set_target_properties(module_test PROPERTIES BUILD_RPATH "${RACK_DIR}")
    add_executable(plugin_load_test tests/plugin_load_test.cpp)
    target_include_directories(plugin_load_test PRIVATE "${RACK_DIR}/include" "${RACK_DIR}/dep/include")
    target_link_libraries(plugin_load_test PRIVATE "${RACK_LIBRARY}" ${CMAKE_DL_LIBS})
    set_target_properties(plugin_load_test PROPERTIES BUILD_RPATH "${RACK_DIR}")
    if(UNIX AND NOT APPLE)
        target_link_options(plugin_load_test PRIVATE -Wl,--allow-shlib-undefined)
    endif()
    add_dependencies(plugin_load_test plugin)
    add_test(NAME plugin_load COMMAND plugin_load_test $<TARGET_FILE:plugin>)
    set(GORGON_DIR "" CACHE PATH "Optional matching Gorgon checkout for real module interop")
    if(GORGON_DIR)
        target_compile_definitions(module_test PRIVATE GORGON_SOURCE="${GORGON_DIR}/rack/src/Gorgon.cpp")
    endif()
    add_test(NAME rack_module COMMAND module_test)
endif()

if(MINGW)
    # Rack does not ship the MinGW C++/GCC/pthread runtime DLLs.
    foreach(target plugin module_test plugin_load_test control_test renderer_test worker_test firmware_test jit_test)
        if(TARGET ${target})
            target_link_options(${target} PRIVATE -static -static-libstdc++ -static-libgcc)
        endif()
    endforeach()
endif()

foreach(target control_test renderer_test worker_test firmware_test jit_test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
if(TARGET module_test)
    if(MSVC)
        target_compile_options(module_test PRIVATE /UNDEBUG)
        target_compile_options(plugin_load_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(module_test PRIVATE -UNDEBUG)
        target_compile_options(plugin_load_test PRIVATE -UNDEBUG)
    endif()
endif()
option(MD_RACK_SANITIZE "Sanitize the adapter and tests (not the emulated firmware)" OFF)
if(MD_RACK_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target rack_adapter control_test renderer_test worker_test firmware_test)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()
